/* 🔔 1,500Hz 超高周波数・緊急アラート音生成 */ function playAlertSound(isTest = false) { try { const audioCtx = new (window.AudioContext || window.webkitAudioContext)(); // 1,500Hzの高音ビープを連続で打つパターン const beepCount = isTest ? 3 : 4; const beepInterval = 0.12; // 短い間隔で連続打鐘 for (let i = 0; i < beepCount; i++) { const startTime = audioCtx.currentTime + (i * beepInterval); const osc = audioCtx.createOscillator(); const gain = audioCtx.createGain(); // 鋭く響くサイン波と1,500Hzの周波数 osc.type = 'sine'; osc.frequency.setValueAtTime(1500, startTime); // 音の立ち上がりと切れ味をはっきりさせる設定 gain.gain.setValueAtTime(0.2, startTime); gain.gain.exponentialRampToValueAtTime(0.001, startTime + 0.08); osc.connect(gain); gain.connect(audioCtx.destination); osc.start(startTime); osc.stop(startTime + 0.08); } } catch (e) { console.warn("音声再生がブロックされました", e); } }
